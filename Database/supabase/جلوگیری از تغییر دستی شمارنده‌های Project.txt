create or replace function block_manual_project_update()
returns trigger as $$
begin
  raise exception 'Project task counters are read-only and auto-managed.';
end;
$$ language plpgsql;

create trigger trg_block_project_update
before update of
  todo_count,
  in_progress_count,
  done_count,
  undefined_count
on projects
for each row
execute function block_manual_project_update();
